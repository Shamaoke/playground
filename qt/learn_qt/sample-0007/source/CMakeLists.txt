
## Требовать CMake версии 4.4 или новее. При использовании более старой
## версии остановить конфигурирование с ошибкой.
## Установить новое поведение политик CMake, введённых в версии 4.4 и ранее.
cmake_minimum_required(VERSION 4.4)

## Объявить проект `sample-0007` версии `0.0.0`.
## Включить поддержку языка C++ (`CXX`), чтобы CMake определил и проверил
## компилятор, а также подготовил правила сборки исходных файлов на C++.
project(sample-0007 VERSION 0.0.0 LANGUAGES CXX)

## Подключить модуль `CTest`, объявляющий опцию `BUILD_TESTING`
## со значением `ON` по умолчанию. При включённой опции модуль вызывает
## `enable_testing()`, разрешая регистрацию тестов для запуска через `ctest`.
## Эту же опцию используем ниже для условного добавления Qt-тестов и Spix.
include(CTest)

## Найти Qt 6 версии не ниже 6.12 с компонентами, необходимыми приложению:
## `Gui` — графическое приложение, окна и события ввода;
## `Qml` — движок QML и взаимодействие QML с объектами C++;
## `Quick` — визуальные элементы Qt Quick и их отображение в `QQuickView`.
## `REQUIRED` останавливает конфигурирование с ошибкой, если подходящая
## версия Qt или любой из перечисленных компонентов не найдены.
## Найденные библиотеки будут подключены к цели приложения ниже.
find_package(
  Qt6 6.12

  COMPONENTS
    Gui
    Qml
    Quick

  REQUIRED
)

## Выполнять поиск тестовых зависимостей и настройку модульных тестов
## только при включённой опции `BUILD_TESTING`.
## Это позволяет конфигурировать обычную сборку без пакетов Spix и Qt Test.
if(BUILD_TESTING)

  ## Найти Spix для управления интерфейсом приложения из интеграционных тестов.
  ## Компонент `Core` предоставляет выполнение команд и XML-RPC-сервер,
  ## а `QtQuick` связывает команды Spix с элементами и событиями Qt Quick.
  ## `CONFIG` задаёт поиск через конфигурационный файл пакета Spix,
  ## установленный вместе с библиотеками, например `SpixConfig.cmake`.
  ## `REQUIRED` останавливает конфигурирование с ошибкой, если пакет
  ## или любой из требуемых компонентов не найден.
  find_package(
    Spix

    CONFIG

    COMPONENTS
      Core
      QtQuick

    REQUIRED
  )

  ## Найти компонент `Test` пакета Qt 6 версии не ниже 6.12.
  ## Он предоставляет средства запуска модульных проверок `StatusProvider`
  ## и сравнения фактических результатов с ожидаемыми через `QCOMPARE`.
  ## `REQUIRED` останавливает конфигурирование с ошибкой, если подходящая
  ## версия Qt или компонент `Test` не найдены.
  ## Найденную библиотеку подключим к тестовой цели через `Qt::Test`.
  find_package(
    Qt6 6.12

    COMPONENTS
      Test

    REQUIRED
  )

  ## Создать исполняемую цель `status-provider-test` для модульных проверок
  ## класса `StatusProvider`. Отдельная тестовая программа позволяет выполнять
  ## эти проверки без запуска графического приложения.
  ## Исходные файлы и зависимости цели задать следующими командами.
  add_executable(status-provider-test)

  ## Включить автоматическую обработку исходников и заголовков цели через `moc`.
  ## Он генерирует метаобъектный код для классов с `Q_OBJECT`:
  ## `StatusProviderTest` и `StatusProvider`. В частности, этот код позволяет
  ## Qt Test обнаруживать и вызывать тестовые методы, объявленные слотами.
  ## Генерацию и включение полученного кода в сборку поручить CMake.
  set_target_properties(status-provider-test PROPERTIES AUTOMOC ON)

  ## Требовать для тестовой цели режим компиляции не ниже C++23.
  ## CMake подбирает необходимые флаги компилятора и сообщает об ошибке,
  ## если требуемый режим недоступен.
  ## `PRIVATE` применяет требование к самой цели, не передавая его
  ## другим целям как условие использования этой цели.
  target_compile_features(status-provider-test PRIVATE cxx_std_23)

  ## Добавить исходные файлы в тестовую цель: `status_provider_test.cc`
  ## содержит проверки и точку входа тестовой программы, а `status_provider.cc` —
  ## реализацию проверяемого класса, используемую также приложением. `PRIVATE`
  ## включает эти файлы в сборку самой цели, не передавая их другим целям
  ## для компиляции при использовании этой цели.
  target_sources(status-provider-test PRIVATE status_provider_test.cc status_provider.cc)

  ## Подключить библиотеку Qt Test, необходимую для выполнения проверок.
  ## Через цель `Qt::Test` получить также пути к заголовкам, определения
  ## препроцессора и необходимые зависимости библиотеки.
  ## `PRIVATE` задаёт зависимость для сборки самой тестовой программы,
  ## не добавляя её в интерфейс использования этой цели.
  target_link_libraries(status-provider-test PRIVATE Qt::Test)

  ## Зарегистрировать в CTest тест с именем `status-provider`, запускающий
  ## программу, собранную из цели `status-provider-test`.
  ## CMake автоматически подставит путь к исполняемому файлу вместо имени цели.
  ## CTest считает тест успешным при коде завершения программы `0`,
  ## а ненулевой код означает провал проверки.
  add_test(NAME status-provider COMMAND status-provider-test)

endif( )

## Создать исполняемую цель `sample-0007` — основное приложение с интерфейсом
## Qt Quick. Цель присутствует в обычной и тестовой сборках.
## Исходные файлы и библиотеки задать следующими командами;
## поддержку Spix подключить ниже при включённой опции `BUILD_TESTING`.
add_executable(sample-0007)

## Включить автоматический запуск `moc` для основной цели приложения.
## Сгенерированный метаобъектный код `StatusProvider` необходим для работы
## класса с `Q_OBJECT` и вызова из QML методов, отмеченных `Q_INVOKABLE`.
## CMake организует генерацию этого кода и его включение в сборку приложения.
set_target_properties(sample-0007 PROPERTIES AUTOMOC ON)

## Требовать для основной цели режим компиляции не ниже C++23.
## CMake подбирает необходимые флаги компилятора и сообщает об ошибке,
## если требуемый режим недоступен. Требование действует и без пресетов.
## `PRIVATE` применяет его к самой цели, не передавая другим целям
## как условие использования этой цели.
target_compile_features(sample-0007 PRIVATE cxx_std_23)

## Добавить исходные файлы основной программы: `main.cc` содержит точку входа,
## встроенное описание интерфейса QML и подключение к нему объекта C++,
## а `status_provider.cc` — реализацию поставщика статусных сообщений.
## `PRIVATE` включает эти файлы в сборку приложения, не передавая их
## другим целям для компиляции при использовании этой цели.
target_sources(sample-0007 PRIVATE main.cc status_provider.cc)

## Подключить к приложению библиотеки `Gui`, `Qml` и `Quick`, найденные выше.
## Их цели предоставляют параметры компоновки, пути к заголовкам, определения
## препроцессора и необходимые зависимости.
## Имена `Qt::...` не содержат номера версии и здесь соответствуют найденному Qt 6.
## `PRIVATE` задаёт зависимости для сборки самого приложения,
## не добавляя их в интерфейс использования этой цели.
target_link_libraries(
  sample-0007

  PRIVATE
    Qt::Gui
    Qt::Qml
    Qt::Quick
)

## Включить управление приложением через Spix только в тестовой сборке.
## При включённой опции `BUILD_TESTING` подключить библиотеки Spix
## и определить макрос для компиляции тестового кода в `main.cc`.
## Это условие зависит от опции тестирования, а не от типа `Debug` или `Release`.
if(BUILD_TESTING)

  ## Подключить адаптер Spix для Qt Quick, необходимый для работы `QtQmlBot`.
  ## Цель `Spix::QtQuick` предоставляет настройки использования библиотеки
  ## и транзитивно подключает ядро Spix, поэтому отдельно указывать его здесь
  ## не требуется. `PRIVATE` задаёт зависимость для самой программы,
  ## не добавляя её в интерфейс использования цели приложения.
  target_link_libraries(sample-0007 PRIVATE Spix::QtQuick)

  ## Передать компилятору макрос `BUILD_TESTING` для исходников приложения.
  ## В `main.cc` условие `#if defined(BUILD_TESTING)` включает заголовки Spix
  ## и код создания бота и XML-RPC-сервера.
  ## Опция CMake сама по себе не создаёт одноимённого макроса C++.
  ## `PRIVATE` ограничивает определение макроса самой целью приложения.
  target_compile_definitions(sample-0007 PRIVATE BUILD_TESTING)

endif( )
